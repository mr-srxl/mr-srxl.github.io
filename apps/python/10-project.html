10 · A small project

Let's bring everything together. This is a small notes app — you can add a note, list all notes, search, and delete. It saves to a JSON file so your notes persist between runs.

Save the following into a file named notes.py, then run it from the terminal with python notes.py. All the techniques come from earlier chapters: functions, dicts, lists, files, user input, and error handling. 
import json
from pathlib import Path

NOTES_FILE = Path("notes.json")


def load_notes():
    """Load notes from disk. Returns an empty list if the file is missing."""
    if not NOTES_FILE.exists():
        return []
    try:
        with NOTES_FILE.open("r", encoding="utf-8") as f:
            return json.load(f)
    except json.JSONDecodeError:
        print("Warning: notes file was corrupted. Starting fresh.")
        return []


def save_notes(notes):
    """Save the notes list back to disk."""
    with NOTES_FILE.open("w", encoding="utf-8") as f:
        json.dump(notes, f, indent=2, ensure_ascii=False)


def add_note(notes):
    text = input("Note: ").strip()
    if not text:
        print("Empty note — skipped.")
        return
    notes.append({"text": text})
    save_notes(notes)
    print("Saved.")


def list_notes(notes):
    if not notes:
        print("No notes yet.")
        return
    for i, note in enumerate(notes, start=1):
        print(f"{i}. {note['text']}")


def search_notes(notes):
    term = input("Search for: ").strip().lower()
    matches = [n for n in notes if term in n["text"].lower()]
    if not matches:
        print("Nothing found.")
        return
    for note in matches:
        print(f"- {note['text']}")


def delete_note(notes):
    list_notes(notes)
    if not notes:
        return
    try:
        index = int(input("Number to delete: ")) - 1
        removed = notes.pop(index)
        save_notes(notes)
        print(f"Deleted: {removed['text']}")
    except (ValueError, IndexError):
        print("Invalid number.")


def main():
    notes = load_notes()
    menu = {
        "1": ("Add a note",    add_note),
        "2": ("List notes",    list_notes),
        "3": ("Search notes",  search_notes),
        "4": ("Delete a note", delete_note),
        "5": ("Quit",          None),
    }

    while True:
        print()
        for key, (label, _) in menu.items():
            print(f"  {key}. {label}")
        choice = input("Choose: ").strip()

        if choice not in menu:
            print("Unknown choice.")
            continue

        _, action = menu[choice]
        if action is None:
            print("Goodbye.")
            break
        action(notes)


if __name__ == "__main__":
    main()

What's going on here

  • Functions keep each feature small and easy to read.
  • Lists hold the notes; each note is a dict with a text field.
  • enumerate gives us clean numbering.
  • A list comprehension filters for search results.
  • JSON + pathlib save and load notes as real files.
  • try / except handles bad input and a corrupted file gracefully.
  • The menu dict maps choices to functions — a neat pattern you'll see everywhere.
  • if __name__ == "__main__": means "only run main() when this file is run directly" — not when it's imported as a module.
